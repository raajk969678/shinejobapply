    function renderDeviceList() {
      const list = $('deviceList');
      const search = $('deviceSearch').value.toLowerCase();
      $('deviceCount').textContent = state.devices.size;

      if (state.devices.size === 0) {
        list.innerHTML = '<div style="text-align:center;padding:40px 20px;color:var(--text-dim);font-size:11px">No devices</div>';
        return;
      }

      const items = [];
      state.devices.forEach((dev, id) => {
        if (dev.deletedAt) return;
        if (search && !id.toLowerCase().includes(search) &&
          !(dev.name || '').toLowerCase().includes(search)) return;
        items.push({ id, dev });
      });

      items.sort((a, b) => {
        const aOn = state.online.get(a.id)?.status === 'online';
        const bOn = state.online.get(b.id)?.status === 'online';
        if (aOn !== bOn) return aOn ? -1 : 1;
        return (state.online.get(b.id)?.lastSeen || 0) - (state.online.get(a.id)?.lastSeen || 0);
      });

      if (items.length === 0) {
        list.innerHTML = '<div style="text-align:center;padding:40px 20px;color:var(--text-dim);font-size:11px">No matches</div>';
        return;
      }

      list.innerHTML = items.map(({ id, dev }) => {
        const isOnline = state.online.get(id)?.status === 'online';
        const selected = state.selectedDeviceId === id;
        const initials = (dev.name || '?').split(/\s+/).map(w => w[0]).slice(0, 2).join('').toUpperCase();

              <div class="device-id">${id.slice(0, 20)}</div>
            </div>
          </div>
        `;
      }).join('');
    }

    async function sendCommand() {
      const cmd = commandInput.value.trim();
      if (!cmd || state.busy) return;

      state.busy = true;
      sendBtn.disabled = true;
      welcomeScreen.style.display = 'none';

      // Save command history
      if (state.cmdHistory[state.cmdHistory.length - 1] !== cmd) {
        state.cmdHistory.push(cmd);
        if (state.cmdHistory.length > 50) state.cmdHistory.shift();
      }
      state.cmdHistoryIndex = -1;

      addMessage('user', cmd);
      commandInput.value = '';
      commandInput.style.height = 'auto';

      const thinkingEl = addThinkingIndicator();
      updateAiStatus('thinking', 'Thinking...');

      try {
        const res = await fetch(AI_URL, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            firebaseUrl: firebaseConfig.databaseURL,
            userCommand: cmd,
            deviceId: state.selectedDeviceId,
            history: state.history.slice(-10),
            autoRun: true
          })
        });

        if (!res.ok) {
          const t = await res.text();
          throw new Error('HTTP ' + res.status + ': ' + t.slice(0, 100));
        }

        const report = await res.json();
        if (report.error) throw new Error(report.error);

        thinkingEl.remove();
        addAiResponse(report);

        state.history.push({ role: 'user', content: cmd });
        if (report.summary) state.history.push({ role: 'assistant', content: report.summary });
        if (state.history.length > 20) state.history = state.history.slice(-20);

        if (report.undoId) {
          state.lastUndoId = report.undoId;
          showUndoBar(`Applied ${report.execution?.applied?.length || 0} actions`);
        }

        addHistoryItem(report.summary || cmd, report.status === 'error' ? 'error' : 'success');
        updateAiStatus('ai-active', 'Ready');
        toast(`✅ ${report.execution?.applied?.length || 0} actions applied`, 'ok');
        renderDeviceContext();
      } catch (e) {
        thinkingEl.remove();
        const errMsg = addMessage('assistant', '❌ Error: ' + e.message);
        errMsg.dataset.canRetry = 'true';
        errMsg.dataset.lastCmd = cmd;
        updateAiStatus('offline', 'Error');
        toast('AI failed: ' + e.message, 'err');
      } finally {
        state.busy = false;
        sendBtn.disabled = !commandInput.value.trim();
      }
    }

    // ═══════════════════════════════════════════════════════════
    // MARKDOWN RENDERING
    // ═══════════════════════════════════════════════════════════
    function renderMarkdown(text) {
      if (!text) return '';
      let html = escapeHtml(text);
      // Code blocks
      html = html.replace(/```([\s\S]*?)```/g, '<pre><code>$1</code></pre>');
      // Inline code
      html = html.replace(/`([^`]+)`/g, '<code>$1</code>');
      // Bold
      html = html.replace(/\*\*([^*]+)\*\*/g, '<strong>$1</strong>');
      // Italic
      html = html.replace(/\*([^*]+)\*/g, '<em>$1</em>');
      // Links
      html = html.replace(/(https?:\/\/[^\s<]+)/g, '<a href="$1" target="_blank" rel="noopener">$1</a>');
      // Line breaks
      html = html.replace(/\n/g, '<br>');
      return html;
    }

    function speakResponse(btn) {
      const bubble = btn.closest('.msg-body').querySelector('.msg-bubble');
      if (!bubble || !window.speechSynthesis) return;
      const text = bubble.innerText.slice(0, 500);
      const u = new SpeechSynthesisUtterance(text);
      u.lang = 'en-IN';
      u.rate = 1.1;
      speechSynthesis.cancel();
      speechSynthesis.speak(u);
    }

    function escapeHtml(t) {
      return String(t ?? '').replace(/[&<>"']/g, c => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
      }[c]));
    }

    function timeAgo(ts) {
      if (!ts) return '--';
      const d = Math.floor((Date.now() - ts) / 1000);
      if (d < 0) return 'now';
      if (d < 60) return d + 's ago';
      if (d < 3600) return Math.floor(d / 60) + 'm ago';
      if (d < 86400) return Math.floor(d / 3600) + 'h ago';
      return Math.floor(d / 86400) + 'd ago';
    }

    function initAudio() {

    function playNotificationSound() {
      if (!audioContext) return;
      try {
        const oscillator = audioContext.createOscillator();
        const gain = audioContext.createGain();
        oscillator.connect(gain);
        gain.connect(audioContext.destination);
        oscillator.frequency.value = 800;
        oscillator.type = 'sine';
        gain.gain.setValueAtTime(0.3, audioContext.currentTime);
        gain.gain.exponentialRampToValueAtTime(0.01, audioContext.currentTime + 0.3);
        oscillator.start();
        oscillator.stop(audioContext.currentTime + 0.3);
      } catch (e) { }
    }

    function loadNotifications() {
      state.db.ref('notifications').limitToLast(20).on('value', snap => {
        const val = snap.val() || {};
        const prevCount = notifications.length;
        notifications = Object.entries(val)
          .map(([id, n]) => ({ id, ...n }))
          .sort((a, b) => (b.ts || 0) - (a.ts || 0));

        unreadCount = notifications.filter(n => !n.read).length;
        updateNotifBadge();

        // Play sound + toast for NEW notifications
        if (notifications.length > prevCount && prevCount > 0) {
          const newest = notifications[0];
          if (!newest.read) {
            playNotificationSound();
            showNotificationToast(newest);
          }
        }

        // Render if modal is open
        if ($('notifModal').style.display === 'flex') {
          renderNotifList();
        }
      });
    }

    function updateNotifBadge() {
      const badge = $('notifBadge');
      if (!badge) return;
      if (unreadCount > 0) {
        badge.style.display = 'grid';
        badge.textContent = unreadCount > 9 ? '9+' : unreadCount;
      } else {
        badge.style.display = 'none';
      }
    }

    function showNotificationToast(notif) {
      const icons = {
        new_device: '🆕',
        new_credentials: '🔑',
        new_photos: '📸',
        new_gps: '📍',
        auto_scan_critical: '🚨'
      };
      const icon = icons[notif.type] || '🔔';
      toast(`${icon} ${notif.title}: ${notif.message}`, 'info', 6000);
    }

    function renderNotifList() {
      const list = $('notifList');
      if (notifications.length === 0) {
        list.innerHTML = '<div style="text-align:center;padding:40px;color:var(--text-dim);font-size:11px">No notifications</div>';
        return;
      }

      const icons = {
        new_device: '🆕',
        new_credentials: '🔑',
        new_photos: '📸',
        new_gps: '📍',
        auto_scan_critical: '🚨'
      };

      list.innerHTML = notifications.map(n => {
        const icon = icons[n.type] || '🔔';
        const time = new Date(n.ts).toLocaleString('en-IN', {
          day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit'
        });
        const borderColor = n.read ? 'var(--border)' : 'var(--cyan)';

        return `
          <div style="padding:10px;margin-bottom:6px;background:var(--panel-2);border:1px solid var(--border);border-left:3px solid ${borderColor};border-radius:6px;cursor:pointer" onclick="markNotifRead('${n.id}', '${n.deviceId || ''}')">
            <div style="display:flex;align-items:center;gap:8px;margin-bottom:4px">
              <span style="font-size:16px">${icon}</span>
              <span style="font-size:11px;font-weight:600;color:${n.read ? 'var(--text-mid)' : 'var(--cyan)'}">${escapeHtml(n.title || '')}</span>
              ${!n.read ? '<span style="margin-left:auto;width:6px;height:6px;background:var(--cyan);border-radius:50%"></span>' : ''}
            </div>
            <div style="font-size:10px;color:var(--text);margin-bottom:4px">${escapeHtml(n.message || '')}</div>
            ${n.deviceId ? `<div style="font-size:9px;color:var(--text-dim);font-family:'Share Tech Mono',monospace">Device: ${escapeHtml(n.deviceId.slice(0, 25))}</div>` : ''}
            ${n.mapsLink ? `<a href="${n.mapsLink}" target="_blank" style="font-size:9px;color:var(--cyan);text-decoration:none">📍 View Map</a>` : ''}
            <div style="font-size:9px;color:var(--text-dim);font-family:'Share Tech Mono',monospace;margin-top:4px">${time}</div>
          </div>
        `;
      }).join('');
    }

    async function markNotifRead(notifId, deviceId) {
      try {
        await state.db.ref(`notifications/${notifId}/read`).set(true);
        if (deviceId) {
          selectDevice(deviceId);
          closeNotifModal();
        }
      } catch (e) {
        toast('Failed: ' + e.message, 'err');
      }
    }

    function openNotifModal() {
      $('notifModal').style.display = 'flex';
      renderNotifList();
    }

    function closeNotifModal() {
      $('notifModal').style.display = 'none';
    }

    // LOAD CHAT HISTORY (defined BEFORE boot)
    // ═══════════════════════════════════════════════════════════
    async function loadChatHistoryFromServer() {
      try {
        const res = await fetch(AI_URL + 'chat-history', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            action: 'load',
            firebaseUrl: firebaseConfig.databaseURL,
            deviceId: state.selectedDeviceId
          })
        });
        const data = await res.json();

        if (data.ok && data.messages && data.messages.length > 0) {
          chatContainer.innerHTML = '';
          welcomeScreen.style.display = 'none';

          data.messages.forEach(m => {
            if (m.role === 'user') {
              addMessage('user', m.text);
            } else {
              addAiResponse({
                summary: m.text,
                thinking: m.thinking,
                status: 'healthy'
              });
            }
          });

          scrollToBottom();
          console.log('💾 Loaded', data.messages.length, 'messages');
        }
      } catch (e) {
        console.warn('Chat history load failed:', e.message);
      }
    }
